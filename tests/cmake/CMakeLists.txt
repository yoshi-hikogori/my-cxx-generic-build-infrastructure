cmake_minimum_required(VERSION 4.5.0)
project(TargetSettingsChecks LANGUAGES CXX)

if(CASE MATCHES "^generic")
	include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/generic-build-infrastructure/CxxGenericBuildInfrastructureTargets.cmake")
	include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/generic-build-infrastructure/CxxGenericBuildInfrastructureTargets.cmake")
	include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/generic-build-infrastructure/CxxGenericBuildInfrastructureCPU.cmake")
	include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/generic-build-infrastructure/CxxGenericBuildInfrastructureInstrumentation.cmake")
else()
	include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/MyImportStdTargetSettings.cmake")
	include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/MyImportStdTargetSettings.cmake")
endif()
add_library(check_target STATIC main.cpp)

if(CASE STREQUAL "generic-platform")
	if(CHECK_ARCH MATCHES "^(aarch64|ARM64|arm64)$")
		set(baseline armv8.2-a)
	else()
		set(baseline x86-64-v3)
	endif()
	cmake_generic_build_infrastructure_apply_cpu_settings(
		TARGET check_target CPU_BASELINE "${baseline}"
	)
	_cmake_generic_build_infrastructure_cpu_options(expected "${baseline}" OFF)
	get_target_property(options check_target COMPILE_OPTIONS)
	if(NOT expected IN_LIST options)
		message(FATAL_ERROR "汎用CPU関数がコンパイル設定を適用していません")
	endif()
	cmake_generic_build_infrastructure_apply_compile_trace(TARGET check_target)
	get_target_property(options check_target COMPILE_OPTIONS)
	if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang" AND NOT "-ftime-trace" IN_LIST options)
		message(FATAL_ERROR "汎用計測関数がClangのtraceを有効にしていません")
	endif()
elseif(CASE STREQUAL "generic-platform-metadata")
	if(CHECK_ARCH MATCHES "^(aarch64|ARM64|arm64)$")
		set(baseline armv8.2-a)
	else()
		set(baseline x86-64-v3)
	endif()
	cmake_generic_build_infrastructure_apply_cpu_settings(TARGET check_target CPU_BASELINE "${baseline}")
	get_cmake_property(cache_before CACHE_VARIABLES)
	foreach(id IN ITEMS x64 x86_64 ARM64 aarch64 ARM64EC "ARM64;x64" "")
		block()
			set(CMAKE_CXX_COMPILER_ARCHITECTURE_ID "${id}")
			set(CMAKE_SIZEOF_VOID_P "")
			set(CMAKE_CXX_BYTE_ORDER "")
			set(CMAKE_SYSTEM_PROCESSOR unrelated)
			cmake_generic_build_infrastructure_apply_cpu_settings(TARGET check_target CPU_BASELINE "${baseline}")
		endblock()
	endforeach()
	block()
		set(CMAKE_CXX_COMPILER_ARCHITECTURE_ID "")
		set(CMAKE_SIZEOF_VOID_P 4)
		set(CMAKE_CXX_BYTE_ORDER BIG_ENDIAN)
		cmake_generic_build_infrastructure_apply_cpu_settings(TARGET check_target CPU_BASELINE "${baseline}")
	endblock()
	get_cmake_property(cache_after CACHE_VARIABLES)
	if(NOT cache_before STREQUAL cache_after)
		message(FATAL_ERROR "標的metadataの変更により追加のコンパイル検査が行われました")
	endif()
	foreach(variable IN LISTS cache_after)
		if(variable MATCHES "^GENERIC_BUILD_INFRASTRUCTURE_(PLATFORM_|GLIBC$)")
			message(FATAL_ERROR "CPU設定だけで独自の命令セット・実行環境検査が行われました")
		endif()
	endforeach()
elseif(CASE STREQUAL "generic-platform-empty")
	get_cmake_property(cache_before CACHE_VARIABLES)
	cmake_generic_build_infrastructure_apply_cpu_settings(TARGET check_target)
	get_cmake_property(cache_after CACHE_VARIABLES)
	if(NOT cache_before STREQUAL cache_after)
		message(FATAL_ERROR "CPU・実行環境の要件なしで追加のコンパイル検査が行われました")
	endif()
elseif(CASE STREQUAL "generic-cpu-missing")
	cmake_generic_build_infrastructure_apply_cpu_settings(TARGET)
elseif(CASE STREQUAL "generic-cpu-invalid")
	cmake_generic_build_infrastructure_apply_cpu_settings(TARGET check_target TARGET_ARCH OFF)
elseif(CASE STREQUAL "generic-trace-missing")
	cmake_generic_build_infrastructure_apply_compile_trace(TARGET)
elseif(CASE STREQUAL "generic-instrumentation-invalid")
	cmake_generic_build_infrastructure_enable_instrumentation(OUTPUT_DIRECTORY relative)
elseif(CASE STREQUAL "generic")
	cmake_generic_build_infrastructure_apply_compile_settings(TARGET check_target)
	get_target_property(options check_target COMPILE_OPTIONS)
	if(NOT options)
		message(FATAL_ERROR "共通のコンパイル設定が適用されていません")
	endif()
	if("${options}" MATCHES "/(EHsc|MP)(;|$)")
		message(FATAL_ERROR "CMakeまたはbuildツールに委譲したオプションが追加されています")
	endif()
elseif(CASE STREQUAL "valid")
	my_import_std_apply_compile_settings(TARGET check_target)
	get_target_property(options check_target COMPILE_OPTIONS)
	if(NOT options)
		message(FATAL_ERROR "コンパイル設定が適用されていません")
	endif()
elseif(CASE STREQUAL "avx2")
	set(CMAKE_SYSTEM_PROCESSOR ARM64)
	set(CMAKE_CXX_COMPILER_ARCHITECTURE_ID unrelated)
	cmake_generic_build_infrastructure_apply_cpu_settings(TARGET check_target ENABLE_AVX2 ON)
	get_target_property(options check_target COMPILE_OPTIONS)
	if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
		set(expected /arch:AVX2)
	else()
		set(expected -mavx2)
	endif()
	if(NOT expected IN_LIST options)
		message(FATAL_ERROR "AVX2のオプションが適用されていません")
	endif()
elseif(CASE STREQUAL "cpu-valid")
	if(CHECK_ARCH MATCHES "^(aarch64|ARM64|arm64)$")
		set(baseline armv8.2-a)
	else()
		set(baseline x86-64-v3)
	endif()
	cmake_generic_build_infrastructure_apply_cpu_settings(TARGET check_target CPU_BASELINE "${baseline}")
	_cmake_generic_build_infrastructure_cpu_options(expected "${baseline}" OFF)
	get_target_property(options check_target COMPILE_OPTIONS)
	foreach(option IN LISTS expected)
		if(NOT option IN_LIST options)
			message(FATAL_ERROR "CPUオプションが適用されていません：${option}")
		endif()
	endforeach()
elseif(CASE STREQUAL "cpu-option-matrix")
	foreach(CMAKE_CXX_COMPILER_ID IN ITEMS GNU Clang MSVC)
		foreach(CMAKE_CXX_COMPILER_FRONTEND_VARIANT IN ITEMS GNU MSVC)
			if((CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND NOT CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "GNU") OR
				(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC" AND NOT CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC"))
				continue()
			endif()
			foreach(baseline IN ITEMS x86-64-v3 armv8.2-a)
				if(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
					if(baseline STREQUAL "x86-64-v3")
						set(expected /arch:AVX2)
					else()
						set(expected /arch:armv8.2)
					endif()
				elseif(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
					set(expected "/clang:-march=${baseline}")
				else()
					set(expected "-march=${baseline}")
				endif()
				_cmake_generic_build_infrastructure_cpu_options(actual "${baseline}" OFF)
				if(NOT actual STREQUAL expected)
					message(FATAL_ERROR "CPUオプションの組み合わせが想定と異なります：想定=${expected}、実際=${actual}")
				endif()
				_cmake_generic_build_infrastructure_cpu_options(actual "${baseline}" ON)
				if(NOT baseline STREQUAL "x86-64-v3")
					if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
						list(APPEND expected /arch:AVX2)
					else()
						list(APPEND expected -mavx2)
					endif()
				endif()
				if(NOT actual STREQUAL expected)
					message(FATAL_ERROR "AVX2併用時のCPUオプションが想定と異なります")
				endif()
			endforeach()
		endforeach()
	endforeach()
elseif(CASE STREQUAL "cpu-incompatible")
	if(CHECK_ARCH MATCHES "^(aarch64|ARM64|arm64)$")
		set(baseline x86-64-v3)
	else()
		set(baseline armv8.2-a)
	endif()
	cmake_generic_build_infrastructure_apply_cpu_settings(TARGET check_target CPU_BASELINE "${baseline}")
elseif(CASE STREQUAL "cpu-avx2-conflict")
	cmake_generic_build_infrastructure_apply_cpu_settings(TARGET check_target CPU_BASELINE armv8.2-a ENABLE_AVX2 ON)
elseif(CASE STREQUAL "cpu-unknown")
	cmake_generic_build_infrastructure_apply_cpu_settings(TARGET check_target CPU_BASELINE OFF)
elseif(CASE STREQUAL "cpu-unsupported")
	set(CMAKE_CXX_COMPILER_ID Unsupported)
	cmake_generic_build_infrastructure_apply_cpu_settings(TARGET check_target CPU_BASELINE x86-64-v3)
elseif(CASE STREQUAL "instrumentation")
	set(MY_IMPORT_STD_ENABLE_AVX2 OFF)
	set(MY_IMPORT_STD_ENABLE_INSTRUMENTATION ON)
	my_import_std_apply_compile_settings(TARGET check_target)
	get_target_property(options check_target COMPILE_OPTIONS)
	if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
		if(NOT "-ftime-trace" IN_LIST options)
			message(FATAL_ERROR "AVX2の有効・無効にかかわらず、Clangのtraceフラグが適用されていません")
		endif()
	elseif("-ftime-trace" IN_LIST options)
		message(FATAL_ERROR "Clang以外のコンパイラにtraceフラグが適用されています")
	endif()
elseif(CASE STREQUAL "missing")
	my_import_std_apply_compile_settings()
elseif(CASE STREQUAL "missing-value")
	my_import_std_apply_compile_settings(TARGET)
elseif(CASE STREQUAL "unknown")
	my_import_std_apply_compile_settings(TARGET check_target TYPO ON)
elseif(CASE STREQUAL "false-unknown")
	my_import_std_apply_compile_settings(TARGET check_target OFF)
elseif(CASE STREQUAL "false-target")
	add_library(OFF STATIC main.cpp)
	my_import_std_apply_compile_settings(TARGET OFF)
elseif(CASE STREQUAL "nonexistent")
	my_import_std_apply_compile_settings(TARGET nonexistent)
elseif(CASE STREQUAL "alias")
	add_library(Check::alias ALIAS check_target)
	my_import_std_apply_compile_settings(TARGET Check::alias)
	get_target_property(options check_target COMPILE_OPTIONS)
	if(NOT options)
		message(FATAL_ERROR "aliasが指す標的にコンパイル設定が適用されていません")
	endif()
elseif(CASE STREQUAL "false-alias")
	add_library(OFF STATIC main.cpp)
	add_library(Check::alias ALIAS OFF)
	my_import_std_apply_compile_settings(TARGET Check::alias)
	get_target_property(options OFF COMPILE_OPTIONS)
	if(NOT options)
		message(FATAL_ERROR "偽と評価される名前のaliasが指す標的にコンパイル設定が適用されていません")
	endif()
elseif(CASE STREQUAL "imported")
	add_library(imported STATIC IMPORTED)
	my_import_std_apply_compile_settings(TARGET imported)
elseif(CASE STREQUAL "interface")
	add_library(interface_target INTERFACE)
	my_import_std_apply_compile_settings(TARGET interface_target)
elseif(CASE STREQUAL "imported-alias")
	add_library(imported STATIC IMPORTED)
	add_library(Check::imported ALIAS imported)
	my_import_std_apply_compile_settings(TARGET Check::imported)
elseif(CASE STREQUAL "interface-alias")
	add_library(interface_target INTERFACE)
	add_library(Check::interface ALIAS interface_target)
	my_import_std_apply_compile_settings(TARGET Check::interface)
elseif(CASE STREQUAL "custom")
	add_custom_target(custom_target)
	my_import_std_apply_compile_settings(TARGET custom_target)
elseif(CASE STREQUAL "empty-extra")
	_cmake_generic_build_infrastructure_validate_target(check check_target "")
elseif(CASE STREQUAL "diagnostic-scope")
	cmake_diagnostic(GET CMD_NON_TARGET_DIRECTIVE before)
	block(SCOPE_FOR VARIABLES DIAGNOSTICS)
		cmake_diagnostic(SET CMD_NON_TARGET_DIRECTIVE FATAL_ERROR)
		set(temporary isolated)
	endblock()
	cmake_diagnostic(GET CMD_NON_TARGET_DIRECTIVE after)
	if(NOT before STREQUAL after OR DEFINED temporary)
		message(FATAL_ERROR "診断設定または変数のスコープが外部に漏れています")
	endif()
elseif(CASE STREQUAL "diagnostic-install")
	cmake_diagnostic(SET CMD_INSTALL_ABSOLUTE_DESTINATION FATAL_ERROR)
	install(FILES main.cpp DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/absolute")
elseif(CASE STREQUAL "diagnostic-non-target")
	cmake_diagnostic(SET CMD_NON_TARGET_DIRECTIVE FATAL_ERROR)
	add_compile_definitions(SHOULD_USE_TARGET_COMMAND)
else()
	message(FATAL_ERROR "不明なテストCASEです：${CASE}")
endif()
